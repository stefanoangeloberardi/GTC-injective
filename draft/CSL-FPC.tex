\subsection{Size Change Termination and Finite Power Condition}

Binary relations compose: if $R \subseteq X \times Y$ and $R' \subseteq Y \times Z$ then $RR' \subseteq X \times Z$ is such that
$xRR'z$ if and only if there is $y \in Y$ s.t. $xRy$ and $yR'z$, which we abbreviate by $xRyR'z$. Relation compositions are associative 
and have the identities $I_X = \Set{(x,x) \mid x \in X}$ as units.

\begin{definition}[Composition of bi-partite relations]\label{def:bi-partiteRelations}
Let $(S,P), (S',P')$ be bipartite relations over $X$; then we define their {\em composition} $(S, P) \comp (S', P') = (S'' \setminus P'',\; P'')$
where
\[S'' = S S \qquad \mbox{and} \qquad P'' = (S P') \cup (P S') \cup (P P') \]
\end{definition}

The composition of bi-partite relations is bi-partite by construction; also, it is associative
but there is no unique identity as $I_X = S \cup P$ holds for different disjoint $S,P$ as soon $X$ has more than one element. 
We observe that composition is compatible with the union and ordinary composition of binary relations: 
let's write $|(S,P)| = S \cup P$, then
\begin{align}\label{eq:unionComp}
|(S,P) \comp (S',P')| = |(S'' \setminus P'', P'')| &= (S''\setminus P'') \cup P'' = SS' \cup  (S P') \cup (P S') \cup (P P') \nonumber \\
& = (S \cup P) (S' \cup P') = |(S,P)| |(S',P')|
\end{align}

%We shall be interested in (finite) sets of bi-partite relations $\biRel = \Set{(S_1,P_1),...,(S_k,P_k)}$ over the same set $X$.
Let $\biRel^*$ be the closure of a set $\biRel$ of bi-partite relations under composition. Notice that the set of bi-partite relations over the finite set $X$
is a subset of $\wp(X^2) \times \wp(X^2)$ which is finite, hence $\biRel^*$ is finite.
The role of a transition graph $\Graph$ with labels in $\biRel$ is to pick certain compositions of bipartite relations in $\biRel^*$. We formalize this notion
by adapting the usual definition of transitive closure of a graph.

\begin{definition}[Tranistive closure of a transition graph]\label{def:transitiveClosure}
We say that  the graph $\Graph^* = (V, \biRel^*, E^*)$ is the {\em transitive closure} of $\Graph = (V, \biRel, E)$ if
$E^* \subseteq V \times \biRel^* \times V$ is the smallest set such that $E \subseteq E^*$ and
\[ (u, (S,P), v), (v, (S',P'), w) \in E^* \Then (u, (S,P) \comp (S',P'), w) \in E^*\]
If $(u, (S,P) \comp (S',P'), w) \in E^*$ for some vertexes $u, w$ then we say that $(S,P)$ and $(S',P')$ are {\em composable} in $\mathcal{G}$; we say
that $(S,P)$ is {\em self-composable} in $\mathcal{G}$ if $(v, (S,P), v) \in E^*$ for some vertex $v$.
\end{definition}

In words, $\Graph^*$ is the transitive closure of the graph $\Graph$ with edges labeled by the composition of the labels of the corresponding edges in $\Graph$. 
As such, it is essentially a category, but for the lack of identities.

Let $\Graph^* = (V, \biRel^*, E^*)$; we abbreviate $(u, (S,P), v) \in E^*$ by %$\edgeG{(S,P)}{u}{v}{\Graph}$ and $(u, (S,P), v) \in E^*$ by 
$\edgeGstar{(S,P)}{u}{v}{\Graph}$.

\begin{definition}[Trace, path and progress]\label{def:TracePath}
Let $\alpha \leq \omega$ be an ordinal, and $\Graph = (V, \biRel, E)$ a transition graph; then:
\begin{enumerate}[i)]
\item a {\em trace} is a map $\tau:\alpha \to X$;
\item a {\em path} is a map $\ppath:\alpha \to E$ such that for all $i + 1 < \alpha$, $\target{\ppath(i)} = \source{\ppath(i+1)}$;
\item a trace $\tau:\alpha + 1 \to X$ is {\em compatible} with a path $\ppath: \alpha \to E$ written $\tau \models \ppath$, if 
	there is $i_0 < \alpha$ such that for all $i_0 \leq i < i+1 < \alpha$,
	if $\lbl{\ppath(i)} = (S,P)$ then $\tau(i) (S \cup P) \tau(i + 1)$, and if $\tau(i) P \tau(i + 1)$ then $i$
	is a {\em progress point} of $\tau$ in $\ppath$.
\end{enumerate}
\end{definition}

We say that $\tau$ is a trace of $\Graph$ if $\tau \models \ppath$ for some path $\ppath$ of $\Graph$.
In our terminology, the path $\ppath$ and the trace $\tau$ correspond to a {\em multipath} $\mathcal{M}$ and a {\em thread} $\textit{th}$ in \cite{Lee1}, respectively.
A trace $\tau:\alpha \to X$, respectively a path $\ppath:\alpha \to E$ are infinite if $\alpha = \omega$, finite otherwise.
Of course, both $\Graph$ and $\Graph^*$ are finite, so that an infinite path $\ppath$ exists in $\Graph$ only if it has a cycle, and therefore there exists
$(S,P)$ which is self-composable in $\Graph^*$.

In the following lemmas, we relate the progress of a trace $\tau$ w.r.t. $\ppath$ to the composition of the bi-partite relations labeling the edges in $\ppath$.

\begin{lemma}\label{lem:P-lemma}
Let $(S,P) = (S_1,P_1)\comp (S_2,P_2)$ and $x (S_1 \cup P_1) y (S_2 \cup P_2) z$, then
\begin{enumerate}[i)]
\item\label{lem:P-lemma-1} $x P_1 y \Then x P z$
\item\label{lem:P-lemma-2} $y P_2 z \Then x P z$
\item\label{lem:P-lemma-3} $x S_1 y \And y S_2 z \Then x S z$
\end{enumerate}
%\[ x P_1 y \myor y P_2 z \Then x P z \]
\end{lemma}

\begin{proof}
By definition of composition, we have $P = S_1P_2 \cup P_1S_2 \cup P_1P_2$. 
Now $x P_1 y \Then x (P_1S_2 \cup P_1P_2) z$ and $yP_2 z \Then x (S_1P_2 \cup P_1P_2) z$ and we are done by definition of $P$, establishing 
(\ref{lem:P-lemma-1}) and (\ref{lem:P-lemma-2}). Part (\ref{lem:P-lemma-3}) is immediate by definition of $S$.
\end{proof}

\begin{definition}[Size-change Termination - SCT]
\label{def:SCT}
The transition graph $\Graph$ satisfies the {\em Size-change Termination} condition 
if for all infinite path $\ppath$ of $\Graph$ there is an infinite trace $\tau$ such that
$\tau \models \ppath$, which has infinitely many progress points.
\end{definition}


\begin{lemma}
\label{lem:SCT}
A transition graph $\Graph$ satisfies the SCT condition if and only if for all self-composable 
$\edgeGstar{(S,P)}{u}{u}{\Graph}$ such that 
$(S,P) \comp (S,P) = (S,P)$ there is some $x \in X$ such that $x P x$.
\end{lemma}

In view of Lemma \ref{lem:SCT}, SCT can be formulated in many equivalent ways, for instance, as Finite Power Condition (FPC):

\begin{definition}[Finite Power Condition - FPC]
\label{def:FPC}
$\Graph$ satisfies the {\em Finite Power Condition} if for all self-composable 
$\edgestar{(S,P)}{u}{u}$ there are some $n > 0 $ and
$x \in X$ such that $x$ progresses to $x$ in $(S,P)^n$.
\end{definition}


To prove the equivalence between SCT and FPC we use a folklore
algebraic property, that all elements of a finite monoid have some positive idempotent power. For $n > 0$, define $(S,P)^n$ as the $n$-times
self-composition of $(S,P)$.

\begin{lemma}[Idempotent Power]
\label{lem:idempotent-power}
For all $(S,P) \in \biRel^*$ there is $n > 0$ such that $(S,P)^n = (S,P)^n \comp (S,P)^n$.
%Let $M$ be a finite monoid and $x \in M$, then for some $n>0$ we have $x^nx^n=x^n$, where $x^n$ is the $n$-times product of $x$ with itself.
\end{lemma}

\begin{proof}
Since $\biRel^*$ is finite, there are only finitely many powers of $(S,P)$, therefore 
for some $a,b>0$ we have $(S,P)^a = (S,P)^{a+b}$. Then for all $c \ge a$ we have $(S,P)^c = (S,P)^{c+b}$,
hence $(S,P)^c = (S,P)^{c+db}$ for all $d>0$. Take the first $d>0$ such that $db \ge a$: then 
$(S,P)^{db} = (S,P)^{db+db} = (S,P)^{db} \comp (S,P)^{db}$. We take $n=db$.
\end{proof}
