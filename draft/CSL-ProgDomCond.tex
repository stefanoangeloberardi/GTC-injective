Let $RR'$ be the ordinary relation composition; we also write $x R y$ for $(x,y) \in R$.

\begin{definition}[Bi-colored relations]\label{def:bi-coloredRelations}
Let $X = \{x_1, ..., x_n\}$ be a finite set. A {\em bi-colored relation} over $X$, is a pair $(S,P)$ such that $S, P \subseteq X \times X$ where $S \cap P = \emptyset$. For any $x,y \in X$, if $x S y$ (resp. $xPy$) we say that $x$ \emph{weakly progresses} (resp. \emph{progresses}) to $y$ in $(S, P)$. 
If $(S,P)$ and $(S',P')$ are bi-colored relations over the same set $X$ then we define the {\em composition} $(S, P) \circ (S', P') = (S'' \setminus P'',\; P'')$
where
\[S'' = S S \qquad \mbox{and} \qquad P'' = (S P') \cup (P S') \cup (P P') \]
\end{definition}

We observe that the composition of bi-colored relations is compatible with the union and ordinary composition of binary relations: let's write 
$|(S,P)| = S \cup P$, then
\begin{align*}
|(S,P) \circ (S',P')| = |(S'' \setminus P'', P'')| &= (S''\setminus P'') \cup P'' = SS' \cup  (S P') \cup (P S') \cup (P P') \\
& = (S \cup P) (S' \cup P') = |(S,P)| |(S',P')|
\end{align*}


\begin{definition}[Transition graph]\label{def:transitionGraph}
Let $\biRel = \{(S_1,P_1),...,(S_n,P_n) \}$ be finite a set of bi-colored relations over the same set $X$. A {\em transition graph} of
$\mathcal{B}$ is a tuple $\Graph =(V, \biRel, E)$ where $V$ is a finite set of vertices, and $E \subseteq V \times \biRel\times V$
is a (finite) set of labeled edges.
\end{definition}

\begin{definition}[Transitive closure graph]\label{def:transClosure}
Given a transition graph $\Graph$ we define its {\em transitive closure} as the graph $\Graph^* = (V, \biRel^*, E^*)$, where
$\biRel^*$ is the closure of $\biRel$ under composition of bi-colored relations and $E^* \subseteq V \times \biRel^* \times V$
is the smallest set such that $E \subseteq E^*$ and
\[ (u, (S,P), v), (v, (S',P'), w) \in E^* \Longrightarrow (u, (S,P) \circ (S',P'), w) \in E^*\]
If $(u, (S,P) \circ (S',P'), w) \in E^*$ for some vertexes $u, w$ then we say that $(S,P)$ and $(S',P')$ are {\em composed} in $\mathcal{G}$; we say
that $(S,P)$ is {\em self-composed} in $\mathcal{G}$ if $(u, (S,P), u) \in E^*$ for some vertex $u$.
\end{definition}
