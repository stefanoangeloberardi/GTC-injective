If $R \subseteq X \times X$ is a binary relation over $X$ then we write $x R y$ for $(x,y) \in R$. 
A relation $R$ is {\em injective} if for all $x,y,z \in X$, $xRy$ and $xRz$ implies that $y = z$.

\begin{lemma}\label{lem:comInjective}
If both $R$ and $R'$ are injective, then $RR'$ is injective.
\end{lemma}

\begin{proof}
If $xRR'y$ and $xRR'z$ then there are $u,v$ such that $xRu$ and $uR'y$, and $xRv$ and $xR'y$. Then $u = v$ by the injectivity of $R$,
which implies that $y = z$ by the injectivity of $R'$.
\end{proof}

\begin{definition}[Bi-colored relations]\label{def:bi-coloredRelations}
Let $X = \{x_1, ..., x_n\}$ be a finite set. A {\em bi-colored relation} over $X$, is a pair $(S,P)$ such that $S, P \subseteq X \times X$ where $S \cap P = \emptyset$. For any $x,y \in X$, if $x S y$ (resp. $xPy$) we say that $x$ \emph{weakly progresses} (resp. \emph{progresses}) to $y$ in $(S, P)$. 
If $(S,P)$ and $(S',P')$ are bi-colored relations over the same set $X$ then we define the {\em composition} $(S, P) \comp (S', P') = (S'' \setminus P'',\; P'')$
where
\[S'' = S S \qquad \mbox{and} \qquad P'' = (S P') \cup (P S') \cup (P P') \]
\end{definition}

It is easy to see \marginpar{Really?} that the composition of bi-colored relations is associative. Also, we observe that it is compatible with the union and ordinary composition of binary relations: let's write 
$|(S,P)| = S \cup P$, then
\begin{align}\label{eq:unionComp}
|(S,P) \comp (S',P')| = |(S'' \setminus P'', P'')| &= (S''\setminus P'') \cup P'' = SS' \cup  (S P') \cup (P S') \cup (P P') \nonumber \\
& = (S \cup P) (S' \cup P') = |(S,P)| |(S',P')|
\end{align}

We say that $(S, P)$ is injective if $|(S, P)| = S \cup P$ is such in the ordinary sense.

\begin{lemma}\label{lem:biRelInjectiveComp}
If both $(S,P)$ and $(S',P')$ are injective, then $(S,P) \comp (S',P')$ is injective.
\end{lemma}

\begin{proof}
By the above we know that $|(S,P) \comp (S',P')| =  |(S,P)| |(S',P')|$ which is ordinary composition of the injective reation $S \cup P$ with the injective
relation $S' \cup P'$, hence it is injective by Lemma \ref{lem:comInjective}.
\end{proof}

We shall be interested in (finite) sets of bi-colored relations $\biRel = \Set{(S_1,P_1),...,(S_n,P_n)}$ over the same set $X$.
Let $\biRel^*$ be the closure of $\biRel$ under composition of bi-colored relations; then we use labeled graphs to select some bi-colored relations
in $\biRel^*$.

\begin{definition}[Transition graph]\label{def:transitionGraph}
Let $\biRel = \Set{(S_1,P_1),...,(S_n,P_n) }$ be a finite set of bi-colored relations over the same set $X$. A {\em transition graph} of
$\biRel$ is a tuple $\Graph =(V, \biRel, E)$ where $V$ is a finite set of vertices, and $E \subseteq V \times \biRel\times V$
is a (finite) set of labeled edges.
\end{definition}

\begin{definition}[Transitive closure graph]\label{def:transClosure}
Given a transition graph $\Graph$ we define its {\em transitive closure} as the graph $\Graph^* = (V, \biRel^*, E^*)$, where
$\biRel^*$ is the closure of $\biRel$ under composition of bi-colored relations and $E^* \subseteq V \times \biRel^* \times V$
is the smallest set such that $E \subseteq E^*$ and
\[ (u, (S,P), v), (v, (S',P'), w) \in E^* \Then (u, (S,P) \comp (S',P'), w) \in E^*\]
If $(u, (S,P) \comp (S',P'), w) \in E^*$ for some vertexes $u, w$ then we say that $(S,P)$ and $(S',P')$ are {\em composed} in $\mathcal{G}$; we say
that $(S,P)$ is {\em self-composed} in $\mathcal{G}$ if $(v, (S,P), v) \in E^*$ for some vertex $v$.
\end{definition}

As $\Graph$ is understood, we omit the subscript in $\edge{(S,P)}{u}{v}$ and $\edgestar{(S,P)}{u}{v}$

If $\edgestar{(S,P)}{v}{v}$ then $\edgestar{(S,P)^n}{v}{v}$ for all $n > 0$ by definition of composition over the edges of $\Graph^*$, where $(S,P)^n$ is the conposition of $(S,P)$ with itslef $n$-times, hence the name.

\begin{definition}[Trace and path]\label{def:TracePath}
Let $\alpha \leq \omega$ be an ordinal, then:
\begin{enumerate}[i)]
\item a {\em trace} is a map $\tau:\alpha \to X$;
\item a {\em value flow} is a map $\theta: \alpha \times X \to \Nat$; (to be moved)
\item a {\em path} is a map $\ppath:\alpha \to V$ such that fo all $h < \alpha$ the triple $(p(h), (S, P), p(h+1))$ 
	belongs to $E$ (is an edge of $\Graph$) for some $(S,P) \in \biRel$;
\item a trace $\tau$ is compatible with a path $\ppath$ (with the same domain $\alpha$), written $\tau \models \ppath$, if whenever
	%$(p(h), (S, P), p(h+1)) \in E$ 
	$\edge{(S,P)}{p(h)}{p(h+1)}$
	we have that $\tau(h) (S \cup P) \tau(h+1)$, and $h$ is a {\em progress point} if $\tau(h) P \tau(h+1)$.
\end{enumerate}
\end{definition}

A trace $\tau:\alpha \to X$, respectively a path $\ppath:\alpha \to V$ are infinite if $\alpha = \omega$, finite otherwise.

Remark: $\Nat$ could be any well-founded set; the role of $\theta$ will be clarified later.

\begin{definition}[Global Trace Condition - GTC]
The transition graph $\Graph$ satisfies the {\em Global Trace Condition} if for all infinite path $\ppath$ of $\Graph$ there is an infinite trace $\tau$ such that
$\tau \models \ppath$ and it has infinitely many progress points.

%$\biRel$ is {\em injective} if for all $(S, P) \in \biRel$ the relation $|(S, P)| = S \cup P$ is injective.
\end{definition}

This is the same as the Size-Change-Termination condition (SCT).

Recall that if $R$ is a binary relation, $\dom{R} = \Set{x \mid \exists y.\; x R y}$.

\begin{definition}[Prograss Domain Condition - PDC]\label{def:PDC}
$\Graph$ satisfies the {\em Prograss Domain Condition} if for all self composed $(S,P) \in \biRel^*$ there are $x, y \in  \dom{S \cup P}$ such that
$x$ progresses to $y$:
\[\forall (S,P) \in \biRel^*, v\in V.\; (v, (S,P), v) \in E^* \Then \exists x, y \in \dom{S \cup P}.\; x P y \]
\end{definition}

\begin{lemma}\label{lem:P-lemma}
Let $(S,P) = (S_1,P_1)\comp (S_2,P_2)$ and $x (S_1 \cup P_1) y (S_2 \cup P_2) z$, then
\begin{enumerate}[i)]
\item\label{lem:P-lemma-1} $x P_1 y \Then x P z$
\item\label{lem:P-lemma-2} $y P_2 z \Then x P z$
\end{enumerate}
%\[ x P_1 y \myor y P_2 z \Then x P z \]
\end{lemma}

\begin{proof}
By definition of composition, we have $P = S_1P_2 \cup P_1S_2 \cup P_1P_2$. 

Now $x P_1 y \Then x (P_1S_2 \cup P_1P_2) z$ and $P_2 z \Then x (S_1P_2 \cup P_1P_2) z$ and we are done by definition of $P$.
\end{proof}

\begin{proposition}
If $\Graph$ satisfies the GTC, it satisfies PDC, not vice versa.
\end{proposition}

\begin{proof}
Suppose that $\edgestar{(S,P)}{u}{u}$; then there exists an infinite path $p$ 
\[p : u \stackrel{(S,P)}{\longrightarrow} u \stackrel{(S,P)}{\longrightarrow} u \stackrel{(S,P)}{\longrightarrow}\cdots\]
such that $p(i) = p(0) = u$ for infinitely many $i < \omega$; let $k > 0$ be the least one such that $p(k) = u$ and 
if $0 < j < k$ then $p(j) \neq u$; w.l.o.g. we also assume
that the the subpaths $\edgestar{(S,P)}{u}{u}$ are the same sequence of edges and labels.

By GTC there is a trace $\tau \models p$ that is infinitely progressing. Let $(S, P) = (S_0,P_0)\comp \cdots \comp (S_{k-1},P_{k-1})$ where the
$(S_i,P_i)$ are the bi-colored relations associated with the edges of the path from $p(0)$ to $p(k)$; hence, there is $h$ with $0 \leq h < k$ such that
$\tau(h)P_h \tau(h+1)$, namely $h$ is a progressing point w.r.t. to $P_h$.

Let $x = \tau(0)$, $z = \tau(h)$, $z' = \tau(h+1)$, and $y = \tau(k)$. Since $\tau \models p$ we know that 
\[x = \tau(0) (S_0 \cup P_0) \tau(1) \cdots \tau(k-1) (S_{k-1} \cup P_{k-1}) \tau(k) = y\]
so that $x \in \dom{(S_0 \cup P_0) \cdots  (S_{k-1} \cup P_{k-1})} = \dom{S \cup P}$ by (\ref{eq:unionComp}). On the other hand, we have assumed that 
$\edgestar{(S,P)}{p(k)}{p(2k)}$ where $p(k) = u = p(2k)$, hence by the same argument we conclude that $y \in \dom{S \cup P}$.

We know that $(S, P) = (S',P') \comp (S_h,P_h) \comp (S'',P'')$, where $(S',P')= (S_0,P_0)\comp \cdots \comp (S_{h-1},P_{h-1})$ and 
$(S'',P'')= (S_{h+1},P_{h+1})\comp \cdots \comp (S_{k-1},P_{k-1})$. By $\tau \models p$ and that $z P_h z'$ we deduce that
\[x (S' \cup P') z P_h z' (S'' \cup P'') y\]
If $(\hat{S},\hat{P}) = (S',P') \comp (S_h,P_h)$ then by (\ref{lem:P-lemma-2}) of Lemma \ref{lem:P-lemma} we have $x \hat{P} z'$; from this and since
$(S,P) = (\hat{S},\hat{P}) \comp (S'', P'')$, by (\ref{lem:P-lemma-1}) of the same Lemma we conclude $x P y$, as desired.
\end{proof}
