\section{Injectivity and Progress Domain Condition}
\label{sec:progressDomainCond}

\input{CSL-injective}

In the previous section, we have seen that the SCT condition for transition graphs and the GCT condition for cyclic proofs coincide up to rephrasing GTC in terms of transition graphs; henceforth, we shall use GTC and SCT as synonymous. We have also proved that the transition graphs associated to cyclic pre-proofs in systems $\CLJIDomega$ and $\CLKIDomega$ are injective. In this section, we formalize a new condition
on transition graphs, which we call {\em Progress Domain Condition}, shortly PDC, and show that, although it is strictly weaker than GCT, it implies GTC in the case of injective graphs. 

\begin{definition}[Prograss Domain Condition - PDC]\label{def:PDC}
$\Graph$ satisfies the {\em Prograss Domain Condition} if for all self-composable $(S,P) \in \biRel^*$ there are $x, y \in  \dom{S \cup P}$ such that
$x$ progresses to $y$:
\[\forall (S,P) \in \biRel^*, v \in V.\;\edgestar{(S,P)}{v}{v} \Then \exists x, y \in \dom{S \cup P}.\; x P y \]
\end{definition}

\begin{proposition}\label{prop:GTCimplPDC}
If $\Graph$ satisfies GTC then it satisfies PDC.
\end{proposition}

\begin{proof}
Let $\edgestar{(S,P)}{v}{v}$ be a self-composable path of $\Graph^*$. Then there is a cycle in $\Graph^*$ and an infinite path
\[\ppath : v \stackrel{(S,P)}{\longrightarrow} v \stackrel{(S,P)}{\longrightarrow} v \stackrel{(S,P)}{\longrightarrow}\cdots\]
where $(S,P) = (S_{0,k-1},P_{0,k-1})$ where $k > 0$ is the length of the cycle. By GTC, there is a trace $\tau = x_0, x_1, \ldots$ such that $\tau\models\ppath$ which is infinitely progressing. Therefore, $\tau$ must be compatible with $\ppath$ since the index $0$, and it must progress in all the finite subpaths $\edgestar{(S,P)}{v}{v}$.

By Proposition \ref{prop:progresing-and-P}, it follows that $x_h P x_k$ for some $0 \leq h < k$; now $x_h \in \dom{P} \subseteq \dom{S\cup P}$, and $x_k \in \dom{S \cup P}$ since
$\tau\models \ppath$. It follows that $\Graph$ satisfies PDC.
\end{proof}

\begin{remark}\label{rem:PDCnotGTC}
The inverse of Proposition \ref{prop:GTCimplPDC} does not hold. Consider $X_0 = \Set{x_1, x_2}$ and $\biRel_0 = \Set{(S_0,P_0)}$ where
$S_0 = \Set{(x_1, x_1)}$ and $P_0 = \Set{(x_2, x_1)}$. Let $\Graph_0 = (\Set{v}, \biRel_0, \Set{\edge{(S_0,P_0)}{v}{v}})$. Then $(S_0,P_0)$
is self-composing, $x_1, x_2 \in \dom{S_0 \cup P_0}$ and $x_2 P_0 x_1$, hence $\Graph_0$ satisfies the PDC.

On the other hand, the infinite path $\ppath_0$ represented in Figure \ref{fig:PDC-counterexample} is such that the only infinite traces compatible with it are
$\tau_0 : x_1 , x_1, \cdots$ and $\tau_1 : x_2, x_1 , x_1, \cdots$, and neither of them is infinitely progressing. \input{CSL-PDC-figure}
\end{remark}

The key point in the counterexample of Remark \ref{rem:PDCnotGTC} is that $\biRel_0$ is not injective. Toward establishing that PDC implies GTC
in case of an injective $\biRel$, we prove the lemma bolow. 

We say that $(S,P)$ is {\em transitive} if $S \cup P$ is such, namely $x(S \cup P)y(S \cup P)z$ implies $x(S \cup P) z$, for all $x, y, z \in X$.

\begin{lemma}\label{lem:transitivity}
Let $\Graph$ be injective and satisfy the PDC.
If $(S,P)$ is self-composing in $\Graph$ and transitive then there exists $x \in X$ such that $xPx$.
\end{lemma}

\begin{proof}
By the hypothesis that $\Graph$ satisfies PDC, there exist $x,y \in \dom{S \cup P}$ such that $xPy$. Since $P \subseteq S \cup P$ we also
have that $x(S \cup P)y$. On the other hand, since $y \in \dom{S \cup P}$, there is $z \in X$ such that $y(S \cup P)z$. From this, by transitivity, it follows that
$x(S \cup P)z$, hence $x = y$ since $(S,P)$ is injective.
\end{proof}

In the proof of Theorem \ref{thr:PDC+injImpliesGTC} below, we use the infinite Ramsey's Theorem. We preliminarily introduce some terminology.
Let $N \subseteq \Nat$, then $[N]^m = \Set{Y \subseteq N \mid |Y| = m}$ can be regarded as the complete (undirected) hypergraph where
all hyperedges $Y \subseteq N$ have the same cardinality $m$ (also called $m$-regular in the literature).
Taking $m = 2$, we have that $[N]^2 = \Set{\Set{x,y} \subseteq N \mid x \neq y}$ is the set of edges of the complete undirected graph whose vertices are in $N$. 

A {\em finite coloring} of $[N]^2$ is a map  
$c:[N]^2 \to \underline{k}$ for some $k > 0$, where $\underline{k} = \Set{1, \ldots , k}$, the set of ``colors'', is the set of the first $k$ positive integers.
A subset $[A]^2 \subseteq [N]^2$ is {\em monocromatic} w.r.t. the coloring $c$ if there is $i \in \underline{k}$ such that  $c(Y) = i$ for all
$Y \in [A]^2$.

%\[c''[Z]^m = \Set{i \mid \exists U \in [Z]^m.\; c(U) = i} \]
%A set $H \subseteq N$ is {\em $c$-homogeneous} if $c''[H]^m$ is a singleton.

\begin{theorem}[Ramsey's Theorem]\label{thr:Ramsey}
For all finite coloring $c$ of $[\Nat]^2$ there exists $A \subseteq \Nat$, $A$ infinite, such that $[A]^2$ is monocromatic with respect to $c$.
\end{theorem}

\begin{proof} See e.g. Theorem 5 in Chapter 1 of \cite{Graham1980}.
\end{proof}

By abuse of language, we say that $c:E \to \underline{k}$ is a finite coloring for any set $E$ of edges.

\begin{corollary}\label{cor:Ramsey}
Let $G = (V, E)$ be a complete undirected graph with infinite denumerable $V$, and let $c: E \to \underline{k}$ be a finite coloring of its edges,
then there exists an infinite $U \subseteq V$ that is a complete subgraph of $G$ and such that for all distinct $u, v \in H$ we have $c(\Set{u,v}) = i$ for some $i \in \underline{k}$.
\end{corollary}

\begin{proof}
As $V$ is denumerable, the graphs $[\Nat]^2$ and $[V]^2$ are isomorphic.
% and the set of edges $E = \Set{ \Set{u, v} \subseteq V \mid u \neq v} = [V]^2$.
Hence,  up to isomorphism, the corollary is a particular case of Theorem \ref{thr:Ramsey}.
\end{proof}


\begin{theorem}[Main Theorem]\label{thr:PDC+injImpliesGTC}
Let $\Graph = (V, \biRel, E)$ be a transition graph with injective $\biRel$; if $\Graph$ satisfies PDC, then it satisfies GTC.
\end{theorem}

be an infinite path of $\Graph = (V, \biRel, E)$, where $v_i = \source{\ppath(i)}$ for $i < \omega$.
Also, $\ppath$ is a path of $\Graph^*$, where for all $0 \leq i < j < \omega$ there is an edge $\edgestar{(S_{i,j}, P_{i,j})}{v_i}{v_j}$, where
$(S_{i,j}, P_{i,j}) = (S_{i}, P_{i}) \comp \cdots \comp (S_{j-1}, P_{j-1})$.

Let $G = (\mathcal{V}, \mathcal{E})$ be the undirected graph formed by the sets
$\mathcal{V} = \Set{(i, v_i) \mid i < \omega}$, which is denumerable, and $\mathcal{E} = \Set{\Set{(i, v_i), (j, v_j)} \mid i < j}$; hence $G$ is complete.
