\section{Progress Domain Condition}
\label{sec:progressDomainCond}

\begin{definition}[Prograss Domain Condition - PDC]\label{def:PDC}
$\Graph$ satisfies the {\em Prograss Domain Condition} if for all self-composable $(S,P) \in \biRel^*$ there are $x, y \in  \dom{S \cup P}$ such that
$x$ progresses to $y$:
\[\forall (S,P) \in \biRel^*, u \in V.\;\edge{(S,P)}{u}{u} \Then \exists x, y \in \dom{S \cup P}.\; x P y \]
\end{definition}

\begin{lemma}\label{lem:P-lemma}
Let $(S,P) = (S_1,P_1)\comp (S_2,P_2)$ and $x (S_1 \cup P_1) y (S_2 \cup P_2) z$, then
\begin{enumerate}[i)]
\item\label{lem:P-lemma-1} $x P_1 y \Then x P z$
\item\label{lem:P-lemma-2} $y P_2 z \Then x P z$
\item\label{lem:P-lemma-2} $x S_1 y \And y S_2 z \Then x S z$
\end{enumerate}
%\[ x P_1 y \myor y P_2 z \Then x P z \]
\end{lemma}

\begin{proof}
By definition of composition, we have $P = S_1P_2 \cup P_1S_2 \cup P_1P_2$. 
Now $x P_1 y \Then x (P_1S_2 \cup P_1P_2) z$ and $yP_2 z \Then x (S_1P_2 \cup P_1P_2) z$ and we are done by definition of $P$.
\end{proof}

\begin{proposition}\label{prop:GTCimplPDC}
If $\Graph$ satisfies GTC then it satisfies PDC.
\end{proposition}

\begin{proof}
Suppose that $\edgestar{(S,P)}{u}{u}$; then there exists an infinite path $\ppath$ \marginpar{to be checked}
\[\ppath : u \stackrel{(S,P)}{\longrightarrow} u \stackrel{(S,P)}{\longrightarrow} u \stackrel{(S,P)}{\longrightarrow}\cdots\]
such that $\source{\ppath(i)} = \source{\ppath(0)} = u$ for infinitely many $i < \omega$; let $k > 0$ be the least one such that $\source{\ppath(k)} = u$ and 
if $0 < j < k$ then $\source{\ppath(j)} \neq u$; w.l.o.g. we also assume that the subpaths $\edgestar{(S,P)}{u}{u}$ of $p$ are the same sequence of edges and labels.

By GTC there is a trace $\tau \models \ppath$ that is infinitely progressing. Let $(S, P) = (S_0,P_0)\comp \cdots \comp (S_{k-1},P_{k-1})$ where the
$(S_i,P_i) \in \biRel$ are the bi-partite relations associated with the edges of the path from $p(0)$ to $p(k)$; hence, there is $h$ with $0 \leq h < k$ such that
$\tau(h)P_h \tau(h+1)$, namely $h$ is a progressing point w.r.t. $P_h$.

Let $x = \tau(0)$, $z = \tau(h)$, $z' = \tau(h+1)$, and $y = \tau(k)$. Since $\tau \models \ppath$ we know that 
\[x = \tau(0) (S_0 \cup P_0) \tau(1) \cdots \tau(k-1) (S_{k-1} \cup P_{k-1}) \tau(k) = y\]
so that $x \in \dom{(S_0 \cup P_0) \cdots  (S_{k-1} \cup P_{k-1})} = \dom{S \cup P}$ by (\ref{eq:unionComp}). On the other hand, we have assumed that 
$\edgestar{(S,P)}{\ppath(k)}{\ppath(2k)}$ where $\ppath(k) = u = \ppath(2k)$, hence by the same argument we conclude that $y \in \dom{S \cup P}$.

We know that $(S, P) = (S',P') \comp (S_h,P_h) \comp (S'',P'')$, where $(S',P')= (S_0,P_0)\comp \cdots \comp (S_{h-1},P_{h-1})$ and 
$(S'',P'')= (S_{h+1},P_{h+1})\comp \cdots \comp (S_{k-1},P_{k-1})$. By $\tau \models p$ and that $z P_h z'$ we deduce that
\[x (S' \cup P') z P_h z' (S'' \cup P'') y\]
If $(\hat{S},\hat{P}) = (S',P') \comp (S_h,P_h)$ then by (\ref{lem:P-lemma-2}) of Lemma \ref{lem:P-lemma} we have $x \hat{P} z'$; from this and 
$(S,P) = (\hat{S},\hat{P}) \comp (S'', P'')$, we conclude that $x P y$  by (\ref{lem:P-lemma-1}) of the same Lemma, as desired.
\end{proof}

\begin{remark}\label{rem:PDCnotGTC}
The inverse of Proposition \ref{prop:GTCimplPDC} does not hold. Consider $X_0 = \Set{x_1, x_2}$ and $\biRel_0 = \Set{(S_0,P_0)}$ where
$S_0 = \Set{(x_1, x_1)}$ and $P_0 = \Set{(x_2, x_1)}$. Let $\Graph_0 = (\Set{v}, \biRel_0, \Set{\edge{(S_0,P_0)}{v}{v}})$. Then $(S_0,P_0)$
is self-composing, $x_1, x_2 \in \dom{S_0 \cup P_0}$ and $x_2 P_0 x_1$, hence $\Graph_0$ satisfies the PD.

On the other hand, the infinite path $\ppath_0$ represented in Figure \ref{fig:PDC-counterexample} is such that the only infinite traces compatible with it are
$\tau_0 : x_1 , x_1, \cdots$ and $\tau_1 : x_2, x_1 , x_1, \cdots$, and neither of them is infinitely progressing. \input{CSL-PDC-figure}
\end{remark}


We say that $(S,P)$ is {\em transitive} if $S \cup P$ is such, namely $x(S \cup P)y(S \cup P)z$ implies $x(S \cup P) z$, for all $x, y, z \in X$.

\begin{lemma}\label{lem:transitivity}
Let $\Graph$ be injective and satisfy the PDC.
If $(S,P)$ is self-composing in $\Graph$ and transitive then there exists $x \in X$ such that $xPx$.
\end{lemma}

\begin{proof}
By the hypothesis that $\Graph$ satisfies PDC, there exist $x,y \in \dom{S \cup P}$ such that $xPy$. Since $P \subseteq S \cup P$ we also
have that $x(S \cup P)y$. On the other hand, since $y \in \dom{S \cup P}$, there is $z \in X$ such that $y(S \cup P)z$. From this, by transitivity, it follows that
$x(S \cup P)z$, hence $x = y$ since $(S,P)$ is injective.
\end{proof}

In the proof of Theorem \ref{thr:PDC+injImpliesGTC} below, we use infinite Ramsey's Theorem. We preliminarily introduce some terminology.
Let $N \subseteq \Nat$, then $[N]^m = \Set{Y \subseteq N \mid |Y| = m}$ can be regarded as the complete (undirected) hypergraph where
all hyperedges $Y \subseteq N$ have the same cardinality $m$ (also called $m$-regular in the literature).\marginpar{where?}
Given a coloring $c: [N]^m \to \Set{1,\ldots, k}$ and $Z \subseteq N$ we define $c''[Z]^m$ as the set of colors of the hyperedges in $[Z]^m$:
\[c''[Z]^m = \Set{i \mid \exists U \in [Z]^m.\; c(U) = i} \]
A set $H \subseteq N$ is {\em $c$-homogeneous} if $c''[H]^m$ is a singleton.

\begin{theorem}[Ramsey's Theorem]\label{thr:Ramsey}
For all $m, k \in \Nat\setminus \Set{0}$ and any coloring $c[\Nat]^m \to \Set{1,\ldots, k}$, there exists an infinite $H \subseteq \Nat$ which is $c$-homogeneous.
\end{theorem}

\begin{corollary}\label{cor:Ramsey}
Let $G = (V, E)$ be a complete undirected graph with infinite denumerable $V$, and let $c: E \to \Set{1,\ldots, k}$ be a finite coloring of its edges,
then there exists an infinite $H \subseteq V$ that is a complete subgraph of $G$ and such that for all distinct $u, v \in H$ we have $c(\Set{u,v}) = i$ for some $i \in \Set{1,\ldots, k}$.
\end{corollary}

\begin{proof}
As $V$ is denumerable, it can be identified with $\Nat$. Then the set of edges $E = \Set{ \Set{u, v} \subseteq V \mid u \neq v} = [V]^2$;
now the corollary is a particular case of Theorem \ref{thr:Ramsey}.
\end{proof}


\begin{theorem}[Main Theorem]\label{thr:PDC+injImpliesGTC}
Let $\Graph = (V, \biRel, E)$ be a transition graph with injective $\biRel$; if $\Graph$ satisfies PDC, then it satisfies GTC.
\end{theorem}

\begin{proof}
Let \[\ppath : v_0 \stackrel{(S_0,P_0)}{\longrightarrow} v_1 \stackrel{(S_1,P_1)}{\longrightarrow} v_2 \stackrel{(S_2,P_2)}{\longrightarrow}\cdots\]
be an infinite path of $\Graph = (V, \biRel, E)$, where $v_i = \source{\ppath(i)}$ for $i < \omega$.
Also, $\ppath$ is a path of $\Graph^*$, where for all $0 \leq i < j < \omega$ there is an edge $\edgestar{(S_{i,j}, P_{i,j})}{v_i}{v_j}$, where
$(S_{i,j}, P_{i,j}) = (S_{i}, P_{i}) \comp \cdots \comp (S_{j-1}, P_{j-1})$.

Let $G = (\mathcal{V}, \mathcal{E})$ be the undirected graph formed by the sets
$\mathcal{V} = \Set{(i, v_i) \mid i < \omega}$, which is denumerable, and $\mathcal{E} = \Set{\Set{(i, v_i), (j, v_j)} \mid i < j}$; hence $G$ is complete.

Note that the edges in $\mathcal{E}$ correspond
to the edges $\edgestar{(S_{i,j}, P_{i,j})}{v_i}{v_j}$ of $\Graph^*$. Since all the $(S_{i,j}, P_{i,j}) \in \biRel^*$, which is a finite set, the mapping
$c\Set{(i, v_i), (j, v_j)} = (S_{i,j}, P_{i,j})$ is a finite coloring of $\mathcal{E}$.

$(S, P) \in \biRel^*$.

injective and satisfies PDC, the Lemma \ref{lem:transitivity} implies that there exists $x \in X$ such that $x P x$. Then it remains to show that there is a trace $\tau$ such that $\tau \models \ppath$ and that whenever $v_i = u$, $\tau(i) = x$: then $x P x$ implies that $\tau$ is infinitely progressing.
